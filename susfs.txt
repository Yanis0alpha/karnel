#ifndef _LINUX_SUSFS_H
#define _LINUX_SUSFS_H

#include <linux/types.h>
#include <linux/stat.h>
#include <linux/path.h>

struct st_susfs_sus_kstat;
struct st_susfs_uname;
struct st_susfs_open_redirect;
struct kstat;

/* SUSFS Definitions */
#define SUSFS_MAGIC 0x53555346
#define CMD_SUSFS_ADD_SUS_PATH_LOOP 0x5001
#define CMD_SUSFS_HIDE_SUS_MNTS_FOR_NON_SU_PROCS 0x5002
#define CMD_SUSFS_ADD_SUS_MAP 0x5003
#define CMD_SUSFS_ENABLE_AVC_LOG_SPOOFING 0x5004

#define INODE_STATE_SUS_PATH (1 << 24)
#define INODE_STATE_SUS_KSTAT (1 << 26)
#define TASK_STRUCT_NON_ROOT_USER_APP_PROC (1 << 28)

#ifndef CL_COPY_MNT_NS
#define CL_COPY_MNT_NS (1 << 20)
#endif
#ifndef CL_ZYGOTE_COPY_MNT_NS
#define CL_ZYGOTE_COPY_MNT_NS (1 << 21)
#endif

/* SUSFS Prototypes */
int susfs_add_sus_kstat(struct st_susfs_sus_kstat* __user user_info);
int susfs_set_uname(struct st_susfs_uname* __user user_info);
int susfs_set_cmdline_or_bootconfig(char* __user user_fake_boot_config);
int susfs_add_open_redirect(struct st_susfs_open_redirect* __user user_info);

#endif /* _LINUX_SUSFS_H */
